/* ============================================================================
   markets.html — prediction markets, measured

   ⚠️ NOTHING ON THIS PAGE IS TYPED IN. Every price, count and date is fetched from
   Polymarket's public API in the visitor's browser at load time. There is no cached
   mirror under /data/ for this page yet, which means two honest consequences that are
   stated on the page rather than hidden: the numbers cannot go stale, and they are not
   reproducible between two visitors. When the /data/ mirror lands, this flips.

   ⚠️ THE CALIBRATION PASS IS OPT-IN ON PURPOSE. It costs one extra request per market
   (the CLOB price history), so an automatic run would fire hundreds of requests at page
   load for a visitor who came to read the board. It is a button.
   ============================================================================ */
(function(){
  "use strict";
  var $ = function(id){ return document.getElementById(id); };
  var esc = function(s){ return String(s==null?"":s).replace(/[&<>"']/g,function(c){
    return ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"})[c]; }); };

  var GAMMA = "https://gamma-api.polymarket.com/markets";
  var CLOB  = "https://clob.polymarket.com/prices-history";

  function stamp(d){
    try{ return d.toISOString().replace("T"," ").slice(0,16) + " UTC"; }
    catch(e){ return "unknown"; }
  }
  function usd(n){
    if(n==null || !isFinite(n)) return "—";
    if(n>=1e9) return "$" + (n/1e9).toFixed(2) + "B";
    if(n>=1e6) return "$" + (n/1e6).toFixed(1) + "M";
    if(n>=1e3) return "$" + Math.round(n/1e3) + "k";
    return "$" + Math.round(n);
  }
  /* Gamma returns JSON-encoded STRINGS for outcomes/outcomePrices/clobTokenIds, not
     arrays. Parsing them defensively is not paranoia — a bare JSON.parse on a null
     field takes the whole page down. */
  function pjson(v){
    if(v==null) return null;
    if(Array.isArray(v)) return v;
    try{ var r = JSON.parse(v); return Array.isArray(r) ? r : null; }catch(e){ return null; }
  }
  function fetchJSON(url){
    return fetch(url, {headers:{"Accept":"application/json"}}).then(function(r){
      if(!r.ok) throw new Error("HTTP " + r.status);
      return r.json();
    });
  }

  /* ---------------- shared tooltip ---------------- */
  var tip = document.createElement("div");
  tip.className = "ddtip";
  document.body.appendChild(tip);
  function showTip(html, ev){
    tip.innerHTML = html; tip.style.display = "block";
    var pad = 14, w = tip.offsetWidth, h = tip.offsetHeight;
    var x = ev.clientX + pad, y = ev.clientY + pad;
    if(x + w > innerWidth - 8) x = ev.clientX - w - pad;
    if(y + h > innerHeight - 8) y = ev.clientY - h - pad;
    tip.style.left = x + "px"; tip.style.top = y + "px";
  }
  function hideTip(){ tip.style.display = "none"; }

  /* ================= SHEET 1 — the open board ================= */
  var board = [], sortKey = "v", sortDir = -1;

  function drawBoard(){
    var q = ($("mkQ").value || "").toLowerCase().trim();
    var rows = board.filter(function(m){ return !q || m.q.toLowerCase().indexOf(q) >= 0; });
    rows.sort(function(a,b){
      var x = a[sortKey], y = b[sortKey];
      if(typeof x === "string") return sortDir * x.localeCompare(y);
      return sortDir * ((x==null?-Infinity:x) - (y==null?-Infinity:y));
    });
    $("mkBody").innerHTML = rows.map(function(m){
      return '<tr><td class="tl"><a href="https://polymarket.com/event/' + esc(m.slug) +
        '" target="_blank" rel="noopener">' + esc(m.q) + '</a></td>' +
        '<td>' + (m.p==null ? "—" : (m.p*100).toFixed(1) + "%") + '</td>' +
        '<td>' + usd(m.v) + '</td>' +
        '<td>' + esc(m.dtxt) + '</td></tr>';
    }).join("");
    $("mkTw").hidden = false;
    $("mkMsg").hidden = rows.length > 0;
    if(!rows.length){ $("mkMsg").hidden = false; $("mkMsg").textContent = "No markets match that filter."; }
  }

  function loadBoard(){
    var n = $("mkN").value;
    $("mkMsg").hidden = false; $("mkMsg").className = "ddmsg";
    $("mkMsg").textContent = "Loading live markets…";
    $("mkTw").hidden = true;
    var url = GAMMA + "?closed=false&active=true&limit=" + encodeURIComponent(n) +
              "&order=volumeNum&ascending=false";
    fetchJSON(url).then(function(raw){
      var arr = Array.isArray(raw) ? raw : (raw && raw.data) || [];
      board = arr.map(function(m){
        var pr = pjson(m.outcomePrices), oc = pjson(m.outcomes);
        /* Only binary Yes/No markets. A multi-outcome market has no single "the price",
           and quietly showing its first leg as if it were one would be a lie. */
        if(!pr || pr.length !== 2 || !oc || oc.length !== 2) return null;
        var p = parseFloat(pr[0]);
        var end = m.endDate ? new Date(m.endDate) : null;
        return {
          q: m.question || m.slug || "(untitled)",
          slug: m.slug || "",
          p: isFinite(p) ? p : null,
          v: typeof m.volumeNum === "number" ? m.volumeNum : parseFloat(m.volume || "0"),
          d: end ? end.getTime() : null,
          dtxt: end ? end.toISOString().slice(0,10) : "—"
        };
      }).filter(Boolean);
      $("mkAsOf").textContent = "fetched " + stamp(new Date());
      var tot = board.reduce(function(s,m){ return s + (m.v||0); }, 0);
      var near = board.filter(function(m){ return m.p!=null && m.p>0.45 && m.p<0.55; }).length;
      $("mkTiles").innerHTML =
        tile("Binary markets", board.length, "of the top " + n + " by volume") +
        tile("Total volume", usd(tot), "lifetime, all shown") +
        tile("Coin flips", near, "priced 45–55%");
      drawBoard();
    }).catch(function(e){
      $("mkTw").hidden = true;
      $("mkMsg").className = "ddmsg err";
      $("mkMsg").innerHTML = "Could not reach Polymarket from your browser (" + esc(e.message) +
        "). This panel reads the API directly, so an ad-blocker, a corporate network or an API " +
        "outage will all look like this. Nothing is cached on our side to fall back to.";
    });
  }

  function tile(k,v,n){
    return '<div class="s"><div class="k">' + esc(k) + '</div><div class="v">' + esc(String(v)) +
           '</div><div class="n">' + esc(n||"") + '</div></div>';
  }

  /* ================= SHEET 2 — the calibration pass ================= */
  /* The domain split reuses THIS run rather than fetching again — one pass, two panels. */
  var lastRun = null;

  /* Wilson score interval. Normal-approximation intervals go outside [0,1] and are
     badly wrong in exactly the buckets we care about most — the 0–10% and 90–100%
     ends, where p is near a boundary and n is small. */
  function wilson(k, n){
    if(!n) return [0,1];
    var z = 1.959964, p = k/n, d = 1 + z*z/n;
    var c = (p + z*z/(2*n)) / d;
    var h = (z * Math.sqrt(p*(1-p)/n + z*z/(4*n*n))) / d;
    return [Math.max(0, c-h), Math.min(1, c+h)];
  }

  /* Small concurrency pool. Polymarket documents 1,000 req/10s on prices-history, so
     8 in flight is nowhere near the limit — the cap is here to be a good citizen and
     to keep a phone from opening 240 sockets at once. */
  function pool(items, size, worker, onTick){
    var i = 0, done = 0, out = new Array(items.length);
    return new Promise(function(resolve){
      function next(){
        if(i >= items.length){ if(done >= items.length) resolve(out); return; }
        var idx = i++;
        worker(items[idx], idx).then(function(r){ out[idx] = r; })
          .catch(function(){ out[idx] = null; })
          .then(function(){ done++; if(onTick) onTick(done, items.length);
            if(done >= items.length) resolve(out); else next(); });
      }
      for(var k = 0; k < Math.min(size, items.length); k++) next();
    });
  }

  /* ================= domain classification =================
     ⚠️ FIXED PRIORITY ORDER, FIRST MATCH WINS, AND IT IS PUBLISHED ON THE PAGE. A market
     tagged both "Politics" and "Crypto" has to land somewhere; picking silently would make
     the politics bucket unfalsifiable. Anything matching nothing lands in "Everything
     else", which is a grab bag and is labelled as one — never a clean category. */
  var DOMAINS = [
    {key:"politics", label:"Politics", pat:/politic|election|trump|biden|congress|senate|president|geopolit|governor|parliament|supreme court/},
    {key:"sports",   label:"Sports",   pat:/sport|nfl|nba|mlb|nhl|soccer|football|basketball|baseball|hockey|tennis|golf|ufc|boxing|olympic|premier league|formula/},
    {key:"crypto",   label:"Crypto",   pat:/crypto|bitcoin|btc|ethereum|solana|dogecoin|stablecoin|defi|altcoin/},
    {key:"other",    label:"Other / untagged", pat:null}
  ];
  function classify(m){
    var tags = Array.isArray(m.tags) ? m.tags : [];
    var hay = tags.map(function(t){
      return String((t && (t.label || t.slug)) || t || "");
    }).join(" ").toLowerCase();
    /* Question text is a FALLBACK ONLY, for markets carrying no tags at all. Letting it
       override a real tag would quietly reclassify markets by their wording. */
    if(!hay) hay = String(m.question || "").toLowerCase();
    for(var i = 0; i < DOMAINS.length; i++){
      if(DOMAINS[i].pat && DOMAINS[i].pat.test(hay)) return DOMAINS[i].key;
    }
    return "other";
  }

  /* ================= logistic recalibration fit =================
     Regress outcome on logit(price). The slope is the whole point:
       slope > 1  prices COMPRESSED toward 50% — long shots UNDERpriced
       slope < 1  prices too EXTREME           — long shots OVERpriced
       slope ~ 1  consistent with honest pricing
     Newton-Raphson on two parameters; the standard error falls out of the inverse
     Hessian, which is the only reason a slope is quotable at these sample sizes. */
  function logitFit(pts){
    var n = pts.length;
    if(n < 25) return null;
    var x = [], y = [];
    for(var i = 0; i < n; i++){
      /* Clip before the logit, or a market quoted at 0.000 sends the design matrix to
         infinity and takes the fit with it. */
      var p = Math.min(0.99, Math.max(0.01, pts[i].p));
      x.push(Math.log(p/(1-p)));
      y.push(pts[i].o);
    }
    var b0 = 0, b1 = 1;
    for(var it = 0; it < 40; it++){
      var g0 = 0, g1 = 0, h00 = 0, h01 = 0, h11 = 0;
      for(var j = 0; j < n; j++){
        var eta = b0 + b1*x[j];
        var mu = 1/(1 + Math.exp(-eta));
        var w = Math.max(1e-9, mu*(1-mu));
        var r = y[j] - mu;
        g0 += r; g1 += r*x[j];
        h00 += w; h01 += w*x[j]; h11 += w*x[j]*x[j];
      }
      var det = h00*h11 - h01*h01;
      if(!isFinite(det) || Math.abs(det) < 1e-12) return null;
      var d0 = ( h11*g0 - h01*g1)/det;
      var d1 = (-h01*g0 + h00*g1)/det;
      b0 += d0; b1 += d1;
      if(!isFinite(b0) || !isFinite(b1) || Math.abs(b1) > 25) return null;
      if(Math.abs(d0) < 1e-9 && Math.abs(d1) < 1e-9) break;
    }
    var H00 = 0, H01 = 0, H11 = 0;
    for(var k = 0; k < n; k++){
      var e2 = b0 + b1*x[k];
      var m2 = 1/(1 + Math.exp(-e2));
      var w2 = Math.max(1e-9, m2*(1-m2));
      H00 += w2; H01 += w2*x[k]; H11 += w2*x[k]*x[k];
    }
    var D = H00*H11 - H01*H01;
    if(!isFinite(D) || D <= 0) return null;
    var se = Math.sqrt(H00/D);
    if(!isFinite(se)) return null;
    return {slope:b1, se:se, lo:b1 - 1.96*se, hi:b1 + 1.96*se, n:n};
  }

  function slopeWords(f){
    if(!f) return "too few markets to fit";
    if(f.lo <= 1 && f.hi >= 1) return "consistent with honest pricing";
    if(f.lo > 1) return "compressed toward 50% — long shots underpriced";
    return "too extreme — long shots overpriced";
  }

  function runCal(){
    var want = parseInt($("calN").value, 10);
    var back = parseInt($("calBack").value, 10);
    var btn = $("calRun");
    btn.disabled = true; btn.textContent = "Running…";
    $("calOut").hidden = true;
    $("calBarW").hidden = false; $("calBar").style.width = "0%";
    $("calMsg").hidden = false; $("calMsg").className = "ddmsg";
    $("calMsg").textContent = "Fetching resolved markets…";

    /* Over-fetch: a good fraction of closed markets are multi-outcome, voided, or have
       no usable history at the lookback, and those are dropped downstream. */
    /* include_tag=true carries Polymarket's own tag objects on each market, so the
       domain split costs ZERO extra requests — it rides this same response. */
    var url = GAMMA + "?closed=true&include_tag=true&limit=" + Math.min(500, want*3) +
              "&order=volumeNum&ascending=false";

    fetchJSON(url).then(function(raw){
      var arr = Array.isArray(raw) ? raw : (raw && raw.data) || [];
      var cands = arr.map(function(m){
        var pr = pjson(m.outcomePrices), oc = pjson(m.outcomes), tk = pjson(m.clobTokenIds);
        if(!pr || pr.length !== 2 || !oc || oc.length !== 2 || !tk || !tk.length) return null;
        /* A cleanly resolved binary market settles at exactly 1/0. Anything else is a
           void, a partial, or still settling — dropped, and counted as dropped. */
        var a = parseFloat(pr[0]), b = parseFloat(pr[1]);
        var outcome = (a === 1 && b === 0) ? 1 : ((a === 0 && b === 1) ? 0 : null);
        if(outcome === null) return null;
        if(!m.endDate) return null;
        return { q: m.question || m.slug, token: tk[0], outcome: outcome,
                 dom: classify(m),
                 end: Math.floor(new Date(m.endDate).getTime()/1000) };
      }).filter(Boolean).slice(0, want);

      if(!cands.length) throw new Error("no cleanly resolved binary markets returned");

      $("calMsg").textContent = "Reading price history for " + cands.length + " markets…";

      return pool(cands, 8, function(c){
        return fetchJSON(CLOB + "?market=" + encodeURIComponent(c.token) +
                         "&interval=max&fidelity=1440").then(function(h){
          var hist = (h && h.history) || [];
          if(!hist.length) return null;
          var target = c.end - back*86400;
          /* Last observation at or before the lookback instant. Using the nearest point
             in either direction would let a post-lookback price leak in, which on a
             market resolving in days is close to leaking the answer. */
          var pick = null;
          for(var i = 0; i < hist.length; i++){
            if(hist[i].t <= target) pick = hist[i]; else break;
          }
          if(!pick) return null;
          var p = parseFloat(pick.p);
          if(!isFinite(p) || p < 0 || p > 1) return null;
          return { p: p, o: c.outcome, q: c.q, dom: c.dom };
        });
      }, function(done, total){
        $("calBar").style.width = Math.round(done/total*100) + "%";
      }).then(function(res){
        var pts = res.filter(Boolean);
        var dropped = cands.length - pts.length;
        if(pts.length < 10) throw new Error("only " + pts.length + " markets survived the filters — too few to bin");
        render(pts, dropped, cands.length, back);
      });
    }).catch(function(e){
      $("calBarW").hidden = true;
      $("calMsg").className = "ddmsg err";
      $("calMsg").innerHTML = "The pass did not finish: " + esc(e.message) + ".";
    }).then(function(){
      btn.disabled = false; btn.textContent = "Run the pass";
    });
  }

  function render(pts, dropped, attempted, back){
    $("calBarW").hidden = true;
    $("calMsg").hidden = true;
    $("calOut").hidden = false;
    $("calAsOf").textContent = "computed " + stamp(new Date());

    var NB = 10, bins = [];
    for(var i = 0; i < NB; i++) bins.push({lo:i/NB, hi:(i+1)/NB, n:0, k:0, sp:0});
    pts.forEach(function(d){
      var idx = Math.min(NB-1, Math.floor(d.p*NB));
      var b = bins[idx]; b.n++; b.k += d.o; b.sp += d.p;
    });
    var live = bins.filter(function(b){ return b.n > 0; });
    live.forEach(function(b){
      b.mean = b.sp/b.n; b.rate = b.k/b.n; b.ci = wilson(b.k, b.n);
    });

    var N = pts.length;
    var ece = live.reduce(function(s,b){ return s + (b.n/N)*Math.abs(b.mean - b.rate); }, 0);
    var brier = pts.reduce(function(s,d){ return s + (d.p-d.o)*(d.p-d.o); }, 0) / N;
    var base = pts.reduce(function(s,d){ return s + d.o; }, 0) / N;

    /* The headline is a sentence, not a metric. ECE is a number for us; "when it said
       70% it happened 68% of the time" is the number for a reader. Pick the populated
       bucket closest to 70% so the sentence is about a real bucket, never a made-up one. */
    var hb = live.slice().sort(function(a,b){
      return Math.abs(a.mean-0.7) - Math.abs(b.mean-0.7); })[0];
    $("calHead").innerHTML = "When these markets said <b>" + Math.round(hb.mean*100) +
      "%</b>, it happened <b>" + Math.round(hb.rate*100) + "%</b> of the time — across " +
      hb.n + " market" + (hb.n===1?"":"s") + " in that bucket, " + N + " overall, priced " +
      back + " day" + (back===1?"":"s") + " before they closed.";

    $("calTiles").innerHTML =
      tile("Markets scored", N, dropped + " dropped of " + attempted) +
      tile("Calibration error", (ece*100).toFixed(1) + " pts", "mean gap, volume-weighted") +
      tile("Brier score", brier.toFixed(3), "lower is better") +
      tile("Base rate", Math.round(base*100) + "%", "share resolving Yes");

    $("calBody").innerHTML = live.map(function(b){
      var gap = b.rate - b.mean;
      var cls = Math.abs(gap) < 0.05 ? "" : (gap > 0 ? "good" : "bad");
      return "<tr><td class=\"tl\">" + Math.round(b.lo*100) + "–" + Math.round(b.hi*100) + "%</td>" +
        "<td>" + b.n + "</td><td>" + (b.mean*100).toFixed(1) + "%</td>" +
        "<td>" + (b.rate*100).toFixed(1) + "%</td>" +
        "<td class=\"" + cls + "\">" + (gap>0?"+":"") + (gap*100).toFixed(1) + "</td></tr>";
    }).join("");

    $("calNote").innerHTML = "Buckets are deciles of the price " + back + " day" +
      (back===1?"":"s") + " before close. Intervals are 95% Wilson. <b>" + dropped +
      "</b> of " + attempted + " markets were dropped — multi-outcome, voided, or no trade " +
      "at the lookback — and a bucket holding a handful of markets is noise wearing a data point.";

    drawFig(live);
    lastRun = {pts: pts, back: back, at: stamp(new Date())};
    renderDomains();
  }

  /* ================= SHEET 4 — the domain split =================
     Same markets, cut by subject. This is the panel that can actually adjudicate the
     compression-vs-longshot disagreement, and it costs one extra query parameter. */
  function binOf(pts, nb){
    var bins = [];
    for(var i = 0; i < nb; i++) bins.push({lo:i/nb, hi:(i+1)/nb, n:0, k:0, sp:0});
    pts.forEach(function(d){
      var b = bins[Math.min(nb-1, Math.floor(d.p*nb))];
      b.n++; b.k += d.o; b.sp += d.p;
    });
    return bins.filter(function(b){ return b.n > 0; }).map(function(b){
      b.mean = b.sp/b.n; b.rate = b.k/b.n; b.ci = wilson(b.k, b.n); return b;
    });
  }
  function eceOf(pts, bins){
    var N = pts.length;
    return bins.reduce(function(s,b){ return s + (b.n/N)*Math.abs(b.mean - b.rate); }, 0);
  }

  function renderDomains(){
    if(!lastRun) return;
    var pts = lastRun.pts;
    $("domMsg").hidden = true;
    $("domOut").hidden = false;
    $("domAsOf").textContent = "computed " + lastRun.at;

    var groups = DOMAINS.map(function(d){
      var sub = pts.filter(function(p){ return p.dom === d.key; });
      return {d: d, pts: sub, fit: logitFit(sub), bins: sub.length ? binOf(sub, 5) : []};
    });

    /* The headline is the politics-vs-everything-else comparison, because that is the
       specific claim in dispute — not a generic "here are four numbers". */
    var pol = groups[0];
    var rest = pts.filter(function(p){ return p.dom !== "politics"; });
    var restFit = logitFit(rest);
    var polE = pol.pts.length ? eceOf(pol.pts, pol.bins) : null;
    var restE = rest.length ? eceOf(rest, binOf(rest, 5)) : null;

    if(pol.pts.length < 25 || rest.length < 25){
      $("domHead").innerHTML = "Not enough markets in this pass to compare politics against the " +
        "rest — <b>" + pol.pts.length + "</b> political and <b>" + rest.length + "</b> non-political. " +
        "Re-run the calibration pass with a larger sample.";
    } else {
      $("domHead").innerHTML = "Politics missed by <b>" + (polE*100).toFixed(1) +
        " points</b> across " + pol.pts.length + " markets; all non-political markets missed by <b>" +
        (restE*100).toFixed(1) + " points</b> across " + rest.length + ". " +
        (pol.fit && restFit
          ? "Slopes " + pol.fit.slope.toFixed(2) + " and " + restFit.slope.toFixed(2) + " — " +
            (pol.fit.lo > restFit.hi || restFit.lo > pol.fit.hi
              ? "their intervals do not overlap, so this pass does separate them."
              : "their intervals overlap, so this pass does <b>not</b> separate them.")
          : "");
    }

    $("domBody").innerHTML = groups.map(function(g){
      var e = g.pts.length ? eceOf(g.pts, g.bins) : null;
      return "<tr><td class=\"tl\">" + esc(g.d.label) + "</td>" +
        "<td>" + g.pts.length + "</td>" +
        "<td>" + (e == null ? "—" : (e*100).toFixed(1) + " pts") + "</td>" +
        "<td>" + (g.fit ? g.fit.slope.toFixed(2) + " ± " + (1.96*g.fit.se).toFixed(2) : "—") + "</td>" +
        "<td class=\"tl\">" + esc(slopeWords(g.fit)) + "</td></tr>";
    }).join("");

    $("domFacets").innerHTML = groups.map(function(g){
      return '<div class="ddfacet"><div class="ft">' + esc(g.d.label) +
        ' <span>' + g.pts.length + ' markets</span></div>' +
        (g.pts.length < 8
          ? '<div class="ddmsg" style="padding:26px 8px">too few to plot</div>'
          : miniFig(g.bins)) + '</div>';
    }).join("");

    Array.prototype.forEach.call($("domFacets").querySelectorAll("circle.dot"), function(c){
      c.addEventListener("mousemove", function(ev){
        var q = c.getAttribute("data-t").split("|");
        showTip("<b>" + esc(q[0]) + "</b><br>" + q.slice(1).map(esc).join("<br>"), ev);
      });
      c.addEventListener("mouseleave", hideTip);
    });

    $("domNote").innerHTML = "Buckets are quintiles here, not deciles — a decile of a subject " +
      "with 40 markets holds four. A slope is fitted only where a subject has at least 25 " +
      "markets. Subjects come from Polymarket&rsquo;s own tags; the keyword list and its priority " +
      "order are in the page source and in <code>llms.txt</code>.";
  }

  /* Small multiples rather than four coloured series on one axis: each facet is a single
     series, so nothing depends on telling four hues apart. */
  function miniFig(bins){
    var W = 300, H = 210, m = {t:10, r:12, b:26, l:34};
    var pw = W-m.l-m.r, ph = H-m.t-m.b;

  function drawFig(bins){
    var W = 680, H = 420, m = {t:16, r:18, b:44, l:70};
    var pw = W - m.l - m.r, ph = H - m.t - m.b;
    var X = function(v){ return m.l + v*pw; };
    var Y = function(v){ return m.t + (1-v)*ph; };
    var s = ['<svg viewBox="0 0 ' + W + ' ' + H + '" role="img" aria-label="Calibration: predicted probability against observed frequency">'];

    for(var g = 0; g <= 10; g += 2){
      s.push('<line class="gr" x1="' + X(g/10) + '" y1="' + Y(0) + '" x2="' + X(g/10) + '" y2="' + Y(1) + '"/>');
      s.push('<line class="gr" x1="' + X(0) + '" y1="' + Y(g/10) + '" x2="' + X(1) + '" y2="' + Y(g/10) + '"/>');
      s.push('<text class="lb" x="' + X(g/10) + '" y="' + (Y(0)+18) + '" text-anchor="middle">' + (g*10) + '%</text>');
      s.push('<text class="lb" x="' + (X(0)-10) + '" y="' + (Y(g/10)+4) + '" text-anchor="end">' + (g*10) + '%</text>');
    }
    s.push('<line class="ax" x1="' + X(0) + '" y1="' + Y(0) + '" x2="' + X(1) + '" y2="' + Y(0) + '"/>');
    s.push('<line class="ax" x1="' + X(0) + '" y1="' + Y(0) + '" x2="' + X(0) + '" y2="' + Y(1) + '"/>');

    /* Reference line is DASHED, and the legend under the chart repeats that dash next to
       its name — so the two marks are told apart by pattern, not by colour alone.
       ⚠️ The in-chart rotated "perfect calibration" caption was removed after the first
       render: a calibrated market hugs the diagonal, so a label ON the diagonal lands on
       top of the data every time. The legend does the same job and cannot collide. */
    s.push('<line class="ref" x1="' + X(0) + '" y1="' + Y(0) + '" x2="' + X(1) + '" y2="' + Y(1) + '"/>');

    var path = bins.map(function(b,i){ return (i?"L":"M") + X(b.mean) + " " + Y(b.rate); }).join(" ");
    s.push('<path class="ser" d="' + path + '"/>');

    bins.forEach(function(b){
      s.push('<line class="wis" x1="' + X(b.mean) + '" y1="' + Y(b.ci[0]) + '" x2="' + X(b.mean) + '" y2="' + Y(b.ci[1]) + '"/>');
      s.push('<circle class="dot" cx="' + X(b.mean) + '" cy="' + Y(b.rate) + '" r="6" ' +
        'data-t="' + esc(Math.round(b.lo*100) + "–" + Math.round(b.hi*100) + "% bucket|" + b.n +
        " markets|said " + (b.mean*100).toFixed(1) + "%|happened " + (b.rate*100).toFixed(1) +
        "%|95% CI " + (b.ci[0]*100).toFixed(0) + "–" + (b.ci[1]*100).toFixed(0) + "%") + '"/>');
    });

    s.push('<text class="lb2" x="' + (m.l + pw/2) + '" y="' + (H-6) + '" text-anchor="middle">What the market said</text>');
    s.push('<text class="lb2" transform="rotate(-90 20 ' + (m.t + ph/2) + ')" x="20" y="' + (m.t + ph/2) + '" text-anchor="middle">What actually happened</text>');
    s.push('</svg>');
    $("calFig").innerHTML = s.join("");

    Array.prototype.forEach.call($("calFig").querySelectorAll("circle.dot"), function(c){
      c.addEventListener("mousemove", function(ev){
        var p = c.getAttribute("data-t").split("|");
        showTip("<b>" + esc(p[0]) + "</b><br>" + p.slice(1).map(esc).join("<br>"), ev);
      });
      c.addEventListener("mouseleave", hideTip);
    });
  }
